# IT-1: 算法链路集成
# =========================================================
def test_alg_pipeline():
    section("IT-1 算法链路集成: get_border → mid_line → angle/dist")

    try:
        from alg import get_border, mid_line, calculate_angle, calculate_dist
    
        # 1) 构造一个规整的四边形轮廓（图像坐标）
        #    左边界 x=5, 右边界 x=595, 上下边界 y=0/800
        keypts = np.array([
            [[5,   0]], [[595,   0]],
            [[595, 800]], [[5, 800]]
        ], dtype=np.int32)
    
        # 2) 预设边界（和轮廓对齐）
        border_left  = ((5,   0), (5,   800))
        border_right = ((595, 0), (595, 800))
    
        # 3) 提取识别边界
        reco_left, reco_right = get_border(
            keypts, *border_left, *border_right
        )
        assert_true(reco_left is not None,  "get_border 返回左边界")
        assert_true(reco_right is not None, "get_border 返回右边界")
    
        if reco_left is None or reco_right is None:
            print("  [SKIP] 后续步骤：边界未识别")
            return
    
        # 4) 计算中线
        reco_mid = mid_line(reco_left, reco_right)
        org_mid  = mid_line(border_left, border_right)
        assert_true(
            reco_mid[0] is not None and reco_mid[1] is not None,
            "mid_line 返回识别中线"
        )
        assert_true(
            org_mid[0] is not None and org_mid[1] is not None,
            "mid_line 返回预设中线"
        )
    
        if None in (reco_mid[0], reco_mid[1], org_mid[0], org_mid[1]):
            print("  [SKIP] 后续步骤：中线为空")
            return
    
        # 5) 计算角度
        try:
            angle = calculate_angle(org_mid, reco_mid)
            assert_true(0.0 <= angle <= 90.0,
                        f"角度在[0,90]范围 (angle={angle:.3f})")
        except Exception as e:
            assert_true(False, "calculate_angle 不抛异常",
                        f"{type(e).__name__}: {e}")
    
        # 6) 计算距离（非垂直线，走正常分支）
        try:
            dist = calculate_dist(
                org_mid, reco_mid,
                border_left, border_right,
                belt_width=2.0
            )
            assert_true(dist >= 0.0,
                        f"距离为非负数 (dist={dist:.4f})")
        except ZeroDivisionError as e:
            assert_true(False, "calculate_dist 不抛除零",
                        "已知缺陷：垂直线/水平线未处理")
        except Exception as e:
            assert_true(False, "calculate_dist 不抛异常",
                        f"{type(e).__name__}: {e}")
    
    except ImportError as e:
        assert_true(False, "导入 alg 模块", str(e))

# IT-2: 配置加载集成
# =========================================================
def test_config_load():
    section("IT-2 配置 → 监控界面 数据加载")

    cfg_path = os.path.join(ROOT, "conf", "config.yaml")
    if not os.path.exists(cfg_path):
        print(f"  [SKIP] 未找到配置文件: {cfg_path}")
        return
    
    try:
        with open(cfg_path, "r", encoding="utf-8") as f:
            cfg = yaml.load(f, Loader=yaml.FullLoader)
        assert_true(isinstance(cfg, dict), "配置文件可解析为 dict")
    
        # 至少一个 belt_x 存在
        belt_keys = [k for k in cfg.keys() if k.startswith("belt_")]
        assert_true(len(belt_keys) > 0,
                    f"存在 belt_N 配置 ({belt_keys})")
    
        # 检查每个 belt 的关键字段
        need_keys = ["url", "belt_width", "angle_alarm",
                     "dist_alarm", "border_left", "border_right"]
        for bk in belt_keys:
            missing = [k for k in need_keys if k not in cfg[bk]]
            assert_true(len(missing) == 0,
                        f"{bk} 包含所有关键字段",
                        f"缺失: {missing}")
    except Exception as e:
        assert_true(False, "配置加载不抛异常", f"{type(e).__name__}: {e}")

# IT-3: 线程信号 → UI 刷新（模拟）
# =========================================================
def test_thread_signal_to_ui():
    section("IT-3 线程信号 → UI 刷新（Qt 无头模式）")

    os.environ.setdefault("QT_QPA_PLATFORM", "offscreen")  # 无窗口运行
    
    try:
        from PyQt5.QtWidgets import QApplication
        from PyQt5.QtCore import QObject, pyqtSignal
    except ImportError as e:
        assert_true(False, "导入 PyQt5", str(e))
        return
    
    app = QApplication.instance() or QApplication(sys.argv)
    
    # 模拟 MyBeltProcess 的信号
    class FakeProcess(QObject):
        done_signal = pyqtSignal(np.ndarray, float, float)
    
    # 模拟 UI 的 refreshFrame
    received = {}
    
    def fake_refresh(frame, distance, angle):
        received["shape"]   = frame.shape
        received["distance"] = distance
        received["angle"]    = angle
    
    proc = FakeProcess()
    proc.done_signal.connect(fake_refresh)
    
    # 发一个假帧
    fake_frame = np.zeros((480, 640, 3), dtype=np.uint8)
    proc.done_signal.emit(fake_frame, 0.25, 3.5)
    
    # 处理事件循环，让信号投递
    app.processEvents()
    
    assert_true(received.get("shape") == (480, 640, 3),
                "信号正确传递 frame", str(received.get("shape")))
    assert_true(received.get("distance") == 0.25,
                "信号正确传递 distance")
    assert_true(received.get("angle") == 3.5,
                "信号正确传递 angle")

# IT-4: 画边界 → 保存 → 重载 一致性
# =========================================================
def test_border_roundtrip():
    section("IT-4 画边界 → 保存配置 → 重新加载 一致性")

    # 直接用 yaml 模拟保存/加载流程（避开 GUI 交互）
    test_cfg = {
        "belt_0": {
            "url": "rtsp://test",
            "snapshot": "",
            "size": {"width": 1920, "height": 1080},
            "belt_width": 180,
            "angle_alarm": 6.0,
            "dist_alarm": 5.0,
            "border_left":  [[10, 20], [10, 900]],
            "border_right": [[600, 20], [600, 900]],
        }
    }
    tmp_file = os.path.join(ROOT, "conf", "_test_cfg_tmp.yaml")
    
    try:
        # 保存
        with open(tmp_file, "w", encoding="utf-8") as f:
            yaml.dump(test_cfg, f)
    
        # 重载
        with open(tmp_file, "r", encoding="utf-8") as f:
            reloaded = yaml.load(f, Loader=yaml.FullLoader)
    
        b = reloaded["belt_0"]
        assert_true(b["border_left"]  == [[10, 20], [10, 900]],
                    "左边界往返一致", str(b["border_left"]))
        assert_true(b["border_right"] == [[600, 20], [600, 900]],
                    "右边界往返一致", str(b["border_right"]))
        assert_true(b["belt_width"] == 180, "皮带宽度往返一致")
        assert_true(b["angle_alarm"] == 6.0, "角度阈值往返一致")
        assert_true(b["dist_alarm"] == 5.0, "距离阈值往返一致")
    except Exception as e:
        assert_true(False, "配置往返不抛异常",
                    f"{type(e).__name__}: {e}")
    finally:
        if os.path.exists(tmp_file):
            os.remove(tmp_file)

# IT-5: 阈值告警逻辑（模拟 refreshFrame 的扩展）
# =========================================================
def test_alarm_logic():
    section("IT-5 告警逻辑（阈值判断）")

    # 模拟 cfg
    cfg = {"angle_alarm": 6.0, "dist_alarm": 5.0}
    
    def alarm_check(angle, dist):
        """模拟未来在 refreshFrame 里应实现的告警"""
        msgs = []
        if angle >= cfg["angle_alarm"]:
            msgs.append(f"角度超限:{angle:.2f}>={cfg['angle_alarm']}")
        if dist >= cfg["dist_alarm"]:
            msgs.append(f"距离超限:{dist:.2f}>={cfg['dist_alarm']}")
        return msgs
    
    assert_true(alarm_check(3.0, 2.0) == [], "正常值不告警")
    assert_true(len(alarm_check(8.0, 2.0)) == 1, "角度超限告警")
    assert_true(len(alarm_check(3.0, 6.0)) == 1, "距离超限告警")
    assert_true(len(alarm_check(8.0, 6.0)) == 2, "双超限告警")

# =========================================================
def test_real_pipeline_optional():
    section("IT-6 (可选) 真实视频 + YOLO 端到端")

    # 想跑这一段，把 ENABLE 改为 True，并保证模型文件存在
    ENABLE = False
    MODEL  = os.path.join(ROOT, "model", "best.pt")
    CFG    = os.path.join(ROOT, "conf",  "config.yaml")
    
    if not ENABLE:
        print("  [SKIP] 默认关闭，如需运行请把 ENABLE 改为 True")
        return
    
    if not os.path.exists(MODEL):
        print(f"  [SKIP] 未找到模型: {MODEL}")
        return
    if not os.path.exists(CFG):
        print(f"  [SKIP] 未找到配置: {CFG}")
        return
    
    try:
        import cv2
        from ultralytics import YOLO
        from alg import get_border, mid_line, calculate_angle, calculate_dist
    
        with open(CFG, "r", encoding="utf-8") as f:
            cfg = yaml.load(f, Loader=yaml.FullLoader)
    
        belt = cfg["belt_0"]
        url  = belt["url"]
    
        cap = cv2.VideoCapture(url)
        assert_true(cap.isOpened(), f"视频流可打开 ({url})")
        ret, frame = cap.read()
        cap.release()
        assert_true(ret, "读取到一帧")
    
        if not ret:
            return
    
        model = YOLO(MODEL)
        res = model.predict(frame, verbose=False)[0]
        masks = res.masks
        assert_true(masks is not None, "YOLO 返回分割 mask")
    
        if masks is None or len(masks.xy) == 0:
            print("  [SKIP] 本帧未检测到目标")
            return
    
        pts = np.asarray(masks.xy[0], dtype=np.int32)
        keypts = cv2.approxPolyDP(pts, 10, True)
    
        bl = tuple(belt["border_left"][0]),  tuple(belt["border_left"][1])
        br = tuple(belt["border_right"][0]), tuple(belt["border_right"][1])
    
        rl, rr = get_border(keypts, *bl, *br)
        assert_true(rl is not None and rr is not None,
                    "端到端：识别出左右边界")
    
        if rl and rr:
            rmid = mid_line(rl, rr)
            omid = mid_line(bl, br)
            if None not in (*rmid, *omid):
                angle = calculate_angle(omid, rmid)
                dist  = calculate_dist(omid, rmid, bl, br,
                                       belt["belt_width"])
                print(f"  端到端结果: angle={angle:.3f}, dist={dist:.4f}")
    
    except Exception as e:
        assert_true(False, "端到端不抛异常",
                    f"{type(e).__name__}: {e}")
        traceback.print_exc()
